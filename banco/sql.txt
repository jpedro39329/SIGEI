-- ============================================================
-- TRIGGER: limite de 3 alunos por PAE
-- ============================================================

DELIMITER $$

CREATE TRIGGER `trg_limite_pae`
BEFORE INSERT ON `associacoes`
FOR EACH ROW
BEGIN
    DECLARE total INT;

    IF NEW.ativo = 1 THEN

        SELECT COUNT(*)
        INTO total
        FROM associacoes
        WHERE id_pae = NEW.id_pae
          AND ativo = 1;

        IF total >= 3 THEN
            SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT =
            'PAE já possui 3 alunos ativos associados';
        END IF;

    END IF;
END$$

DELIMITER ;
