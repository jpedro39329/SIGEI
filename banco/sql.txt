        FOREIGN KEY (id_pae)
        REFERENCES usuarios_pae(id_pae)
        ON UPDATE CASCADE

DELIMITER $$

CREATE TRIGGER trg_limite_pae
BEFORE INSERT ON associacoes
FOR EACH ROW
BEGIN
    DECLARE qtd INT;

    IF NEW.ativo = 1 THEN

        SELECT COUNT(*)
        INTO qtd
        FROM associacoes
        WHERE id_pae = NEW.id_pae
          AND ativo = 1;

        IF qtd >= 3 THEN
            SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT = 'O PAE já possui 3 alunos ativos.';
        END IF;

    END IF;
END$$

DELIMITER ;
