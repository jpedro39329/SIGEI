-- Cria o banco de dados
CREATE DATABASE IF NOT EXISTS `sigei` 
CHARACTER SET utf8mb4 
COLLATE utf8mb4_general_ci;

-- Seleciona o banco de dados

-- --------------------------------------------------------
-- Estrutura para tabela `admin`
-- --------------------------------------------------------
CREATE TABLE `admin` (
  `id_admin` int(11) NOT NULL,
  `nome` varchar(150) NOT NULL,
  `cpf` char(11) NOT NULL,
  `senha` varchar(255) NOT NULL,
  `data_cadastro` datetime DEFAULT current_timestamp()
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

-- --------------------------------------------------------
-- Estrutura para tabela `alunos`
-- --------------------------------------------------------
CREATE TABLE `alunos` (
  `id_aluno` int(11) NOT NULL,
  `nome` varchar(150) NOT NULL,
  `cpf` char(11) DEFAULT NULL,
  `ra` varchar(20) DEFAULT NULL,
  `data_nascimento` date DEFAULT NULL,
  `descricao_deficiencia` varchar(255) DEFAULT NULL,
  `descricao_cuidados` text DEFAULT NULL,
  `nome_responsavel` varchar(150) DEFAULT NULL,
  `cpf_responsavel` char(11) DEFAULT NULL,
  `foto_arquivo` varchar(255) DEFAULT NULL,
  `status_aprovacao` enum('PENDENTE','APROVADO','REPROVADO','ARQUIVADO') DEFAULT 'PENDENTE',
  `motivo_reprovacao` text DEFAULT NULL,
  `id_escola` int(11) NOT NULL,
  `id_usuario_escola` int(11) NOT NULL,
  `data_cadastro` datetime DEFAULT current_timestamp()
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

-- --------------------------------------------------------
-- Estrutura para tabela `associacoes`
-- --------------------------------------------------------
CREATE TABLE `associacoes` (

-- --------------------------------------------------------
-- Estrutura para tabela `educacao_especial`

-- --------------------------------------------------------
-- Estrutura para tabela `empresas`
-- --------------------------------------------------------
CREATE TABLE `empresas` (

DELIMITER $$
CREATE TRIGGER `trg_limite_pae` BEFORE INSERT ON `associacoes` FOR EACH ROW BEGIN
    DECLARE total INT;
    IF NEW.ativo = 1 THEN
        SELECT COUNT(*) INTO total FROM associacoes
        WHERE id_pae = NEW.id_pae AND ativo = 1;
        IF total >= 3 THEN
            SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT = 'PAE já possui 3 alunos ativos associados';
        END IF;
    END IF;
END$$
DELIMITER ;
